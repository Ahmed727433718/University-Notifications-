import android.content.BroadcastReceiver;
import android.content.Context;
import android.content.Intent;

public class AlarmReceiver extends BroadcastReceiver {
    @Override
    public void onReceive(Context context, Intent intent) {
        final Context app = context.getApplicationContext();
        final PendingResult result = goAsync();
        new Thread(new Runnable() {
            @Override public void run() {
                try {
                    Scheduler.scheduleNext(app);
                    Refresher.refreshAndNotify(app);
                } finally {
                    result.finish();
                }
            }
        }).start();
    }
}

package app.studybrief;

import java.time.Instant;
import java.time.LocalDate;
import java.time.ZoneId;
import java.time.temporal.ChronoUnit;
import java.util.ArrayList;
import java.util.Collections;
import java.util.Comparator;
import java.util.List;
import java.util.Locale;

/** Pure logic: which events matter and how to describe them. */
final class Brief {
    private Brief() {}

    private static final String[] QUIZ = {"quiz", "exam", "midterm", "كويز", "اختبار", "امتحان", "امتحـان"};
    private static final String[] DUE = {"due", "deadline", "closes", "assign", "تسليم", "واجب", "مهمة", "مهمه", "يغلق", "إغلاق", "اغلاق", "موعد"};
    private static final String[] OPENS = {"opens", "يفتح", "فتح "};

    private static boolean has(String title, String[] words) {
        String t = title.toLowerCase(Locale.ROOT);
        for (String w : words) if (t.contains(w)) return true;
        return false;
    }

    static boolean isQuiz(String t) { return has(t, QUIZ); }
    static boolean isDue(String t) { return has(t, DUE); }
    static boolean isOpening(String t) { return has(t, OPENS); }

    static String icon(String t) {
        if (isQuiz(t)) return "📝";
        if (isDue(t)) return "📌";
        return "📅";
    }

    static LocalDate dateOf(Event e, ZoneId zone) {
        return Instant.ofEpochMilli(e.start).atZone(zone).toLocalDate();
    }

    static String time(Event e, ZoneId zone) {
        if (e.allDay) return "طوال اليوم";
        return String.format(Locale.ROOT, "%02d:%02d",
                Instant.ofEpochMilli(e.start).atZone(zone).getHour(),
                Instant.ofEpochMilli(e.start).atZone(zone).getMinute());
    }

    static List<Event> sorted(List<Event> evs) {
        List<Event> c = new ArrayList<>(evs);
        Collections.sort(c, new Comparator<Event>() {
            @Override public int compare(Event a, Event b) { return Long.compare(a.start, b.start); }
        });
        return c;
    }

    static List<Event> onDay(List<Event> evs, LocalDate day, ZoneId zone) {
        List<Event> out = new ArrayList<>();
        for (Event e : sorted(evs)) if (dateOf(e, zone).equals(day)) out.add(e);
        return out;
    }

    /** Quizzes/exams within 2 days, deadlines today or tomorrow. "Opens" events are not urgent. */
    static List<Event> important(List<Event> evs, LocalDate today, ZoneId zone) {
        List<Event> out = new ArrayList<>();
        for (Event e : sorted(evs)) {
            if (isOpening(e.title)) continue;
            long days = ChronoUnit.DAYS.between(today, dateOf(e, zone));
            if (days < 0) continue;
            if (isQuiz(e.title) && days <= 2) out.add(e);
            else if (isDue(e.title) && days <= 1) out.add(e);
        }
        return out;
    }

    static String dayLabel(LocalDate d, LocalDate today) {
        long n = ChronoUnit.DAYS.between(today, d);
        if (n == 0) return "اليوم";
        if (n == 1) return "بكرة";
        return "بعد " + n + " أيام";
    }

    static String line(Event e, ZoneId zone) {
        String extra = e.course.isEmpty() ? "" : " (" + e.course + ")";
        return icon(e.title) + " " + time(e, zone) + " — " + e.title + extra;
    }

    static String describe(Event e, LocalDate today, ZoneId zone) {
        String extra = e.course.isEmpty() ? "" : " (" + e.course + ")";
        return icon(e.title) + " " + dayLabel(dateOf(e, zone), today) + " " + time(e, zone) + " — " + e.title + extra;
    }

    /** Text for the morning notification. */
    static String notificationBody(List<Event> evs, LocalDate today, ZoneId zone) {
        List<Event> imp = important(evs, today, zone);
        List<Event> todayList = onDay(evs, today, zone);
        StringBuilder sb = new StringBuilder();
        if (!imp.isEmpty()) {
            sb.append("⚠️ مهم\n");
            for (Event e : imp) sb.append(describe(e, today, zone)).append('\n');
            sb.append('\n');
        }
        if (todayList.isEmpty()) {
            sb.append("ما في شي اليوم 🎉");
        } else {
            sb.append("اليوم\n");
            for (Event e : todayList) sb.append(line(e, zone)).append('\n');
        }
        return sb.toString().trim();
    }
}

package app.studybrief;

import java.time.LocalDate;
import java.time.LocalDateTime;
import java.time.ZoneId;
import java.time.ZoneOffset;
import java.time.format.DateTimeFormatter;
import java.util.ArrayList;
import java.util.HashMap;
import java.util.List;
import java.util.Map;

/** Minimal iCalendar reader for Moodle exports (VEVENT with DTSTART / SUMMARY / CATEGORIES). */
final class Ics {
    private Ics() {}

    private static final DateTimeFormatter DT = DateTimeFormatter.ofPattern("yyyyMMdd'T'HHmmss");

    static List<Event> parse(String text, ZoneId local) {
        String[] raw = text.replace("\r\n", "\n").replace("\r", "\n").split("\n");
        List<String> lines = new ArrayList<>();
        for (String r : raw) {
            if (!r.isEmpty() && (r.charAt(0) == ' ' || r.charAt(0) == '\t') && !lines.isEmpty()) {
                int last = lines.size() - 1;
                lines.set(last, lines.get(last) + r.substring(1)); // unfold continuation line
            } else {
                lines.add(r);
            }
        }

        List<Event> out = new ArrayList<>();
        boolean inEvent = false;
        boolean hasStart = false;
        boolean allDay = false;
        long start = 0;
        String title = "";
        String course = "";

        for (String ln : lines) {
            if (ln.equals("BEGIN:VEVENT")) {
                inEvent = true;
                hasStart = false;
                allDay = false;
                title = "";
                course = "";
                continue;
            }
            if (ln.equals("END:VEVENT")) {
                if (inEvent && hasStart) out.add(new Event(start, allDay, title, course));
                inEvent = false;
                continue;
            }
            if (!inEvent) continue;

            int colon = ln.indexOf(':');
            if (colon < 0) continue;
            String head = ln.substring(0, colon);
            String value = ln.substring(colon + 1);
            String[] parts = head.split(";");
            String name = parts[0].toUpperCase();
            Map<String, String> params = new HashMap<>();
            for (int i = 1; i < parts.length; i++) {
                int eq = parts[i].indexOf('=');
                if (eq > 0) params.put(parts[i].substring(0, eq).toUpperCase(), parts[i].substring(eq + 1));
            }

            if (name.equals("DTSTART")) {
                try {
                    long[] r = parseDt(params, value, local);
                    start = r[0];
                    allDay = r[1] == 1;
                    hasStart = true;
                } catch (RuntimeException ignored) {
                    // skip events with a date we cannot read
                }
            } else if (name.equals("SUMMARY")) {
                title = unescape(value);
            } else if (name.equals("CATEGORIES")) {
                course = unescape(value);
            }
        }
        return out;
    }

    private static long[] parseDt(Map<String, String> params, String raw, ZoneId local) {
        String v = raw.trim();
        if ("DATE".equals(params.get("VALUE")) || v.length() == 8) {
            LocalDate d = LocalDate.parse(v.substring(0, 8), DateTimeFormatter.BASIC_ISO_DATE);
            return new long[]{d.atStartOfDay(local).toInstant().toEpochMilli(), 1};
        }
        if (v.endsWith("Z")) {
            LocalDateTime ldt = LocalDateTime.parse(v.substring(0, v.length() - 1), DT);
            return new long[]{ldt.toInstant(ZoneOffset.UTC).toEpochMilli(), 0};
        }
        ZoneId zone = local;
        String tzid = params.get("TZID");
        if (tzid != null) {
            try {
                zone = ZoneId.of(tzid.replace("\"", ""));
            } catch (RuntimeException e) {
                zone = local;
            }
        }
        LocalDateTime ldt = LocalDateTime.parse(v, DT);
        return new long[]{ldt.atZone(zone).toInstant().toEpochMilli(), 0};
    }

    private static String unescape(String s) {
        return s.replace("\\n", " ").replace("\\N", " ").replace("\\,", ",")
                .replace("\\;", ";").replace("\\\\", "\\").trim();
    }
}

package app.studybrief;

import java.io.ByteArrayOutputStream;
import java.io.IOException;
import java.io.InputStream;
import java.net.HttpURLConnection;
import java.net.URL;

final class Net {
    private Net() {}

    /** Error messages here must never contain the URL (it holds a private token). */
    static String get(String url) throws IOException {
        HttpURLConnection c = (HttpURLConnection) new URL(url).openConnection();
        try {
            c.setConnectTimeout(7000);
            c.setReadTimeout(8000);
            c.setRequestProperty("User-Agent",
                    "Mozilla/5.0 (Linux; Android 14; Mobile) AppleWebKit/537.36 (KHTML, like Gecko) Chrome/126.0 Mobile Safari/537.36");
            c.setRequestProperty("Accept", "text/calendar,text/plain,*/*;q=0.8");
            int code = c.getResponseCode();
            if (code != 200) throw new IOException("HTTP " + code);
            try (InputStream in = c.getInputStream(); ByteArrayOutputStream out = new ByteArrayOutputStream()) {
                byte[] buf = new byte[8192];
                int n;
                while ((n = in.read(buf)) > 0) out.write(buf, 0, n);
                return out.toString("UTF-8");
            }
        } finally {
            c.disconnect();
        }
    }
}

import android.app.Notification;

package app.studybrief;

import android.content.Context;

import java.time.LocalDate;
import java.util.List;

/** Blocking work: call from a background thread only. */
final class Refresher {
    private Refresher() {}

    static boolean refresh(Context c) {
        String url = Store.url(c);
        if (url.isEmpty()) return false;
        try {
            String body = Net.get(url);
            List<Event> evs = Ics.parse(body, Store.zone());
            Store.saveEvents(c, evs, System.currentTimeMillis());
            return true;
        } catch (Exception e) {
            String msg = e.getMessage();
            Store.setError(c, e.getClass().getSimpleName() + (msg == null ? "" : ": " + msg));
            return false;
        }
    }

    /** Fetch, then post the morning notification (uses cached data if the fetch failed). */
    static void refreshAndNotify(Context c) {
        boolean ok = refresh(c);
        List<Event> evs = Store.loadEvents(c);
        if (!ok && evs.isEmpty()) {
            Notifier.show(c, "📚 شو عندك اليوم", "تعذّر تحديث التقويم. افتح التطبيق وتأكد من الرابط والإنترنت.");
            return;
        }
        String body = Brief.notificationBody(evs, LocalDate.now(), Store.zone());
        if (!ok) body += "\n\n(تعذّر التحديث، هاي آخر بيانات محفوظة)";
        Notifier.show(c, "📚 شو عندك اليوم", body);
    }
}

package app.studybrief;

import android.app.AlarmManager;
import android.app.PendingIntent;
import android.content.Context;
import android.content.Intent;

import java.time.ZonedDateTime;

final class Scheduler {
    private Scheduler() {}

    /** Arms one alarm for the next occurrence of the chosen hour. The receiver re-arms it each day. */
    static void scheduleNext(Context c) {
        AlarmManager am = c.getSystemService(AlarmManager.class);
        if (am == null) return;
        ZonedDateTime now = ZonedDateTime.now();
        ZonedDateTime next = now.toLocalDate().atTime(Store.hour(c), 0).atZone(now.getZone());
        if (!next.isAfter(now.plusMinutes(1))) next = next.plusDays(1);

        PendingIntent pi = PendingIntent.getBroadcast(c, 1, new Intent(c, AlarmReceiver.class),
                PendingIntent.FLAG_IMMUTABLE | PendingIntent.FLAG_UPDATE_CURRENT);
        // Inexact (needs no special permission); may arrive a few minutes late when the phone is idle.
        am.setAndAllowWhileIdle(AlarmManager.RTC_WAKEUP, next.toInstant().toEpochMilli(), pi);
    }
}

package app.studybrief;

import android.content.Context;
import android.content.SharedPreferences;

import java.time.ZoneId;
import java.util.ArrayList;
import java.util.List;

/** Private on-device storage. The calendar link never leaves the phone except to fetch the calendar. */
final class Store {
    private Store() {}

    private static SharedPreferences p(Context c) {
        return c.getSharedPreferences("brief", Context.MODE_PRIVATE);
    }

    static String url(Context c) { return p(c).getString("url", ""); }
    static void setUrl(Context c, String u) { p(c).edit().putString("url", u).apply(); }

    static int hour(Context c) { return p(c).getInt("hour", 7); }
    static void setHour(Context c, int h) { p(c).edit().putInt("hour", h).apply(); }

    static long fetchedAt(Context c) { return p(c).getLong("fetchedAt", 0L); }

    static String lastError(Context c) { return p(c).getString("error", ""); }
    static void setError(Context c, String e) { p(c).edit().putString("error", e == null ? "" : e).apply(); }

    static void saveEvents(Context c, List<Event> evs, long fetchedAt) {
        StringBuilder sb = new StringBuilder();
        for (Event e : evs) {
            sb.append(e.start).append('\t').append(e.allDay ? 1 : 0).append('\t')
              .append(clean(e.title)).append('\t').append(clean(e.course)).append('\n');
        }
        p(c).edit().putString("events", sb.toString()).putLong("fetchedAt", fetchedAt).putString("error", "").apply();
    }

    static List<Event> loadEvents(Context c) {
        List<Event> out = new ArrayList<>();
        String raw = p(c).getString("events", "");
        if (raw.isEmpty()) return out;
        for (String line : raw.split("\n")) {
            String[] f = line.split("\t", -1);
            if (f.length < 4) continue;
            try {
                out.add(new Event(Long.parseLong(f[0]), f[1].equals("1"), f[2], f[3]));
            } catch (NumberFormatException ignored) {
                // skip a damaged line
            }
        }
        return out;
    }

    static ZoneId zone() { return ZoneId.systemDefault(); }

    private static String clean(String s) { return s.replace('\t', ' ').replace('\n', ' ').replace('\r', ' '); }
}
